cmake_minimum_required(VERSION 3.24)
project(animejanai_amd_windows LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Linux: configure native/linux-ort with AJI_NATIVE_AMD=ON")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(AJI_SOURCE "" CACHE PATH "Companion inference checkout (perf-specialized-gpu-20261014)")
set(AJI_ORT_ROOT "" CACHE PATH "Microsoft.ML.OnnxRuntime.DirectML package")
set(AJI_DML_ROOT "" CACHE PATH "Microsoft.AI.DirectML package")
if(NOT EXISTS "${AJI_SOURCE}/src/aji_dml.cpp")
  message(FATAL_ERROR "AJI_SOURCE must name the companion inference checkout")
endif()
add_library(aji SHARED "${AJI_SOURCE}/src/aji_dispatch.cpp" "${AJI_SOURCE}/src/aji_conf.cpp")
target_include_directories(aji PRIVATE "${AJI_SOURCE}/include")
add_library(aji_migraphx SHARED "${AJI_SOURCE}/src/aji_dml.cpp" "${AJI_SOURCE}/src/aji_conf.cpp"
  "${AJI_SOURCE}/src/onnx_shape.cpp" "${AJI_SOURCE}/src/onnx_flow.cpp")
target_include_directories(aji_migraphx PRIVATE "${AJI_SOURCE}/include" "${AJI_SOURCE}/src"
  "${AJI_ORT_ROOT}/build/native/include" "${AJI_DML_ROOT}/include" "${CMAKE_CURRENT_SOURCE_DIR}/../linux-ort")
target_compile_definitions(aji_migraphx PRIVATE AJI_MIGRAPHX_INTEROP=1)
target_link_libraries(aji_migraphx PRIVATE d3d12 d3d11 dxgi d3dcompiler)
include(build.cmake)
aji_add_amd_executor(aji_migraphx)
# No CUDA language, NVIDIA SDK, TensorRT or NVIDIA driver needed by this build.
